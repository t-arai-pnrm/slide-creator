{/* 手で作った原稿。/create-slide ができたら、そちらで作り直す */}
{/* ノートの <p> は 1 行で書く。中で改行すると、MDX が中の文を段落として <p> で包み、<p> の中に <p> ができる */}

{/* コードの { } を式として読ませないよう、HTML の文字列で持つ */}
{/* 枠に 2.5cqi の文字（DESIGN.md の code）で収まるよう、閾値の式は「閾値」と書いて短くする。実際の式はノートで話す */}
export const cond = '<span class="-k">@container</span> (<span class="-s">閾値</span> &lt;= inline-size)'
export const scatteredCode = [`._layout    ${cond}`, `._thumbnail ${cond}`, `._title     ${cond}`].join('\n')

{/* 印の行（.-mark）はブロックで描くので、その後ろに改行を置くと空の行が 1 つ増える。印の行だけ改行を付けない */}
export const flagLines = [
	['._title {', false],
	['  font-size: 小さい文字;', false],
	['  <span class="-k">@container</span> <span class="-p">style(--_density: --spacious)</span> {', true],
	['    font-size: 大きい文字;', false],
	['  }', false],
	['}', false],
]
export const flagCode = flagLines.map(([text, mark]) => (mark ? `<span class="-mark">${text}</span>` : `${text}\n`)).join('')

<Slide layout="section" chapter="01">
	<Fragment slot="title">サイズクエリの閾値が散らばる</Fragment>
	<p slot="notes">まず、サイズクエリだけで書いたときに何が困るかを見る。</p>
</Slide>

<Slide chapter="01 — Problem">
	<Fragment slot="title">同じ閾値が、子要素の数だけ並ぶ</Fragment>
	<figure class="code-figure">
		<figcaption class="code-label">CSS</figcaption>
		<pre class="code-block" set:html={scatteredCode} />
	</figure>
	<p slot="notes">記事カードで、サムネイルを横に並べ、正方形にし、題の文字を大きくする。変えたい子要素ごとのルールにサイズクエリを書くと、同じcalc()の閾値が子の数だけ並ぶ。</p>
</Slide>

<Slide layout="statement">
	<Fragment slot="title">1か所でも直し忘れると、<br />デザインにない状態ができる</Fragment>
	<p>横並びなのに、題の文字は小さいまま。</p>
	<p slot="notes">閾値をカスタムプロパティに入れて共有できればよいが、サイズクエリの条件でvar()を使う書き方は、2026年11月の時点で対象のブラウザすべてでは使えない。</p>
</Slide>

<Slide layout="section" chapter="02">
	<Fragment slot="title">スタイルクエリで、判定を1か所にまとめる</Fragment>
	<p slot="notes">ここからが本題。幅を判定した結果を、カスタムプロパティで子孫に伝える。</p>
</Slide>

<Slide chapter="02 — Mechanism">
	<Fragment slot="title">幅を判定する場所と、読む場所を分ける</Fragment>
	<ol class="flow">
		<li>
			<div class="flow-item -strong">
				<span class="flow-label">1. 判定する場所</span>
				<code class="flow-code">._layout</code>
				<span class="flow-text">サイズクエリで幅を見る</span>
			</div>
			<span class="flow-arrow" aria-hidden="true">→</span>
		</li>
		<li>
			<div class="flow-item">
				<span class="flow-label">2. 伝えるフラグ</span>
				<code class="flow-code">--_density</code>
				<span class="flow-text">幅が広いと立てる</span>
			</div>
			<span class="flow-arrow" aria-hidden="true">→</span>
		</li>
		<li>
			<div class="flow-item">
				<span class="flow-label">3. 読む場所</span>
				<code class="flow-code">style()</code>
				<span class="flow-text">子孫が値を読む</span>
			</div>
		</li>
	</ol>
	<p slot="notes">サイズクエリを書くのは、フラグを宣言する._layoutのルールだけ。フラグをルートに置かないのは、コンテナが自分の幅を条件にできないため。</p>
</Slide>

<Slide chapter="02 — Mechanism">
	<Fragment slot="title">子孫のルールには、閾値が出てこない</Fragment>
	<figure class="code-figure">
		<figcaption class="code-label">CSS</figcaption>
		<pre class="code-block" set:html={flagCode} />
	</figure>
	<p slot="notes">狭い幅の見た目はクエリなしで書き、広い幅の見た目だけをstyle()の中に書く。名前を付けない@container style()は親を問い合わせ、継承したフラグも読める。</p>
</Slide>

<Slide chapter="02 — Mechanism">
	<Fragment slot="title">閾値は3か所から1か所になる</Fragment>
	<div class="compare">
		<div class="compare-item">
			<span class="flow-label">Before / サイズクエリだけ</span>
			<span class="compare-value">閾値 × 3</span>
			<p>子要素ごとに同じ条件を書く</p>
		</div>
		<div class="compare-item -strong">
			<span class="flow-label">After / フラグを読む</span>
			<span class="compare-value">閾値 × 1</span>
			<p>判定は1か所、ほかは値を読むだけ</p>
		</div>
	</div>
	<p slot="notes">要素の見た目は、その要素のルールの中にまとまったまま。閾値の1か所化と両立できる。</p>
</Slide>

<Slide layout="statement">
	<Fragment slot="title">ホバーとフォーカスも、フラグにして子孫に配れる</Fragment>
	<p><code>:has()</code>で子のリンクの状態を受け、<code>--_has-hocus: --true</code>を立てる。</p>
	<p slot="notes">記事一覧の行は、押せるのが題のリンクだけで、見た目を変えたいのは日付と矢印。子の状態を:has()で親に上げてから、フラグで子孫に配る。</p>
</Slide>

<Slide layout="section" chapter="03">
	<Fragment slot="title">フラグの決まりと、対応状況</Fragment>
	<p slot="notes">最後に、書き間違いを防ぐ決まりと、使えるブラウザを確認する。</p>
</Slide>

<Slide chapter="03 — Rules">
	<Fragment slot="title">値は<code>--</code>で始まる識別子に、初期値はクエリの外に</Fragment>
	<div class="compare">
		<div class="compare-item">
			<span class="flow-label">避ける</span>
			<span class="compare-value"><code>1</code> / <code>true</code></span>
			<p>書き間違えてもエラーにならず、条件が成り立たないだけ</p>
		</div>
		<div class="compare-item -strong">
			<span class="flow-label">そろえる</span>
			<span class="compare-value"><code>--true</code> / <code>--false</code></span>
			<p>CSSのキーワードとぶつからない</p>
		</div>
	</div>
	<p slot="notes">初期値を書かないと、外側のコンポーネントで同じ名前のフラグが立っていたときに、その値を継承してしまう。範囲の比較はSafariが対応するまで使わない。</p>
</Slide>
